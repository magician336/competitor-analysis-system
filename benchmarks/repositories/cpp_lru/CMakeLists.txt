cmake_minimum_required(VERSION 3.16)
project(coderadar_bench_010 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(lru_checks checks.cpp)
if(MSVC)
  target_compile_options(lru_checks PRIVATE /W4 /WX)
else()
  target_compile_options(lru_checks PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME lru_behavior COMMAND lru_checks)
